> **For agentic workers:** Implement this plan task-by-task.

**Goal:** Rebuild the adult (tutor / parent) shell and pages along option B:
- 3 quick buttons (Chấm bài · Giao bài · Ghi buổi học);
- 5 sections (Hôm nay · Bài tập · Buổi học · Tiến bộ · Kế hoạch) plus Hồ sơ học sinh;
- a Bài tập page with 4 tabs (drafts grouped by Unit);
- a Tiến bộ page with an overview and a chart;
- "Việc nên làm" and "Việc khác" only on Hôm nay.

**Architecture:**
- Presentation only. No SQL and no new API besides one more number in an existing read-only route.
- The left column stays data-driven (`lib/shell/nav.ts` + `lib/shell/paths.ts`, both pure and unit-tested).
- Tabs are plain links between the existing pages, so every old URL still opens and lights the right section.
- New figures (overview tiles, chart points, draft groups) are pure functions over data the pages already load.

**Tech Stack:** Next.js App Router (server components + a few client components), TypeScript strict, Tailwind, Vitest (`renderToStaticMarkup` for UI tests), Playwright e2e in CI.

**Spec:** `docs/superpowers/specs/2026-11-26-ia-huong-b-design.md` (read it first; the table "Đường dẫn" there is the source of truth for which page lights which section).

```ts
/**
 * The staff sidebar as data (pure, unit-tested). Hướng B (26/11, docs/superpowers/specs/2026-11-26-ia-huong-b-design.md): three quick
 * buttons for the daily work, five sections, then Hồ sơ học sinh. No track branching: the same model for every child.
 */
import { aiSettingsCopy, levelCopy, purgePage, readingPage, shellCopy, targetsPage, tutorAsksCopy, weakPage } from "@/lib/copy/vi";

export const SECTION_KEYS = ["desk", "work", "sessions", "progress", "plan", "profile"] as const;
export type SectionKey = (typeof SECTION_KEYS)[number];
export const QUICK_KEYS = ["grade", "assign", "note"] as const;
export type QuickKey = (typeof QUICK_KEYS)[number];
export type NavStatus = "live" | "soon";
export type NavRole = "tutor" | "parent";
export type NavFlags = { revisionsReady: boolean };
export type NavLink = { key: string; label: string; href: string };
export type NavQuick = { key: QuickKey; label: string; href: string };
export type NavSection = { key: SectionKey; label: string; status: NavStatus; lane: string | null; href: string | null; children: NavLink[] };
export type NavModel = { quick: NavQuick[]; sections: NavSection[]; parent: NavLink[] };

export type SectionDef = {
  key: SectionKey;
  status: NavStatus;
  lane: string | null;
  root: (learnerId: string) => string;
  children: (learnerId: string, flags: NavFlags) => NavLink[];
};

const L = (id: string) => `/learners/${id}`;
export const learnerQuery = (id: string) => `?learner=${id}`;

export const QUICK: readonly { key: QuickKey; href: (id: string) => string }[] = [
  { key: "grade", href: (id) => `${L(id)}/finish` },
  { key: "assign", href: (id) => `${L(id)}/assign` },
  { key: "note", href: (id) => `${L(id)}/notes/new` },
];

export const SECTIONS: readonly SectionDef[] = [
  { key: "desk", status: "live", lane: null, root: (id) => `/${learnerQuery(id)}`, children: () => [] },
  { key: "work", status: "live", lane: null, root: (id) => `${L(id)}/assigned`, children: () => [] },
  { key: "sessions", status: "live", lane: null, root: (id) => `${L(id)}/notes`, children: () => [] },
  {
    key: "progress", status: "live", lane: null, root: (id) => `${L(id)}/progress`,
    children: (id) => [
      { key: "results", label: shellCopy.progressResults, href: `${L(id)}/progress` },
      { key: "knowledge", label: shellCopy.progressKnowledge, href: `${L(id)}/knowledge` },
    ],
  },
  { key: "plan", status: "live", lane: null, root: (id) => `${L(id)}/curricula`, children: () => [] },
  {
    key: "profile", status: "live", lane: null, root: (id) => `${L(id)}/levels`,
    children: (id) => [
      { key: "levels", label: levelCopy.link, href: `${L(id)}/levels` },
      { key: "weak", label: weakPage.link, href: `${L(id)}/weak` },
      { key: "targets", label: targetsPage.link, href: `${L(id)}/targets` },
      { key: "reading", label: readingPage.link, href: `${L(id)}/reading/new` },
      { key: "tutorAsks", label: tutorAsksCopy.link, href: `${L(id)}/tutor-asks` },
    ],
  },
];

export function buildNav(role: NavRole, learnerId: string | null, flags: NavFlags): NavModel {
  const sections = SECTIONS.map((s): NavSection => {
    const live = s.status === "live" && learnerId !== null;
    return {
      key: s.key,
      label: shellCopy.sections[s.key],
      status: s.status,
      lane: s.lane,
      href: s.key === "desk" ? (learnerId ? s.root(learnerId) : "/") : live && learnerId ? s.root(learnerId) : null,
      children: live && learnerId ? s.children(learnerId, flags) : [],
    };
  });
  const quick: NavQuick[] = learnerId ? QUICK.map((q) => ({ key: q.key, label: shellCopy.quick[q.key], href: q.href(learnerId) })) : [];
  const parent: NavLink[] =
    role === "parent"
      ? [
          { key: "aiSettings", label: aiSettingsCopy.navLink, href: `/settings/ai${learnerId ? learnerQuery(learnerId) : ""}` },
          ...(learnerId ? [{ key: "delete", label: purgePage.link, href: `${L(learnerId)}/delete` }] : []),
        ]
      : [];
  return { quick, sections, parent };
}
```

```ts
import { type BadgeMemo, shouldFetch } from "@/lib/shell/badgeMemo";

export type BadgeCounts = { count: string | null; work: string | null };
type Memo = BadgeMemo & BadgeCounts;
const NONE: BadgeCounts = { count: null, work: null };
/** One remembered pair per child (this tab). */
const memos = new Map<string, Memo>();
const listeners = new Set<() => void>();
let lastPath: string | null = null;
const notify = () => listeners.forEach((l) => l());
const subscribe = (l: () => void) => {
  listeners.add(l);
  return () => void listeners.delete(l);
};
/** Stable snapshot object per memo, so useSyncExternalStore does not loop. */
const snapshot = (learnerId: string): BadgeCounts => memos.get(learnerId) ?? NONE;

/** Test hook: forget what this tab remembered. */
export function resetBadge(): void {
  memos.clear();
  lastPath = null;
  notify();
}
/** A page that already counted (Hôm nay, Việc khác) hands the numbers over so the route is not asked again. Browser only. */
export function seedBadge(learnerId: string, c: BadgeCounts): void {
  if (typeof window === "undefined") return;
  memos.set(learnerId, { learnerId, at: Date.now(), count: c.count, work: c.work });
}

/**
 * The two numbers, read AFTER the page shows (the page never waits for them). Unknown / network error → both null: never a false
 * number. Same 30 s memo and "re-read after a decision page" rule as before (lib/shell/badgeMemo.ts).
 */
export function useBadgeCounts(learnerId: string | null): BadgeCounts {
  const pathname = usePathname() ?? "/";
  const counts = useSyncExternalStore(subscribe, () => (learnerId ? snapshot(learnerId) : NONE), () => NONE);
  useEffect(() => {
    if (!learnerId) return;
    const left = lastPath;
    lastPath = pathname;
    if (!shouldFetch(memos.get(learnerId) ?? null, learnerId, Date.now(), left)) return;
    const ctl = new AbortController();
    (async () => {
      try {
        const res = await fetch(`/api/learners/${learnerId}/pending-count`, { signal: ctl.signal, cache: "no-store" });
        if (!res.ok) throw new Error(String(res.status));
        const body = (await res.json()) as { count?: unknown; work?: unknown };
        const s = (v: unknown) => (typeof v === "string" ? v : null);
        memos.set(learnerId, { learnerId, at: Date.now(), count: s(body.count), work: s(body.work) });
        notify();
      } catch {
        if (!ctl.signal.aborted) {
          memos.delete(learnerId);
          notify();
        }
      }
    })();
    return () => ctl.abort();
  }, [learnerId, pathname]);
  return counts;
}
```

```ts
export function groupDraftsByUnit(drafts: readonly DraftListItem[], statusOf: Readonly<Record<string, UnitStatus>> | null): DraftGroup[] {
  const groups = new Map<string, DraftListItem[]>();
  for (const d of drafts) {
    const k = unitKeyOf([], d.title) ?? OTHER;
    groups.set(k, [...(groups.get(k) ?? []), d]);
  }
  const keys = [...groups.keys()].sort((a, b) => (a === OTHER ? 1 : b === OTHER ? -1 : unitOrder(a) - unitOrder(b) || a.localeCompare(b)));
  const isUnit = (k: string) => /^U\d+$/.test(k);
  const closed = (k: string) => statusOf?.[k] === "done" || statusOf?.[k] === "skipped";
  const next = keys.find((k) => isUnit(k) && !closed(k)) ?? null;
  const ordered = next ? [next, ...keys.filter((k) => k !== next)] : keys;
  return ordered.map((k) => ({
    key: k,
    unitKey: k === OTHER ? null : k,
    status: statusOf?.[k] ?? null,
    next: k === next,
    drafts: [...(groups.get(k) ?? [])].sort(byTitle),
  }));
}
```

```ts
/**
 * Tiến bộ › Tổng quan (hướng B, 26/11): four numbers, the first-try chart, the weakest / strongest knowledge items. Pure, adults only.
 * Same measure as the rest of Kết quả: counted works (first attempts) only; a % needs at least one decided question; unknown = null.
 * "4 tuần" = the 28 Vietnam days up to today, compared with the 28 days before.
 */
import { addDays, todayVn } from "@/lib/inquiries/dates";
import { RESULT_RULES } from "./actions";
import { add, pct, ratio, type TagAgg, type Tally, type WorkScore } from "./score";

export type MainUnits = { title: string; currentCode: string | null; currentTitle: string | null; done: number; total: number };
export const OVERVIEW_DAYS = 28;
export const CHART_MAX = 20;
export type ChartPoint = { instanceId: string; title: string; unitKey: string | null; day: string; pct: number; correct: number; decided: number };
export type Overview = {
  sent: { now: number; before: number };
  firstTry: { now: number | null; before: number | null; diff: number | null };
  waiting: { works: number; oldestDays: number | null };
  main: MainUnits | null;
  points: ChartPoint[];
  weakest: TagAgg[];
  strongest: TagAgg[];
};

const ZERO: Tally = { correct: 0, decided: 0, pending: 0 };
const dayOf = (iso: string) => todayVn(new Date(iso));

export function buildOverview(works: readonly WorkScore[], tags: readonly TagAgg[], main: MainUnits | null, todayIso: string): Overview {
  const startNow = addDays(todayIso, -(OVERVIEW_DAYS - 1));
  const startBefore = addDays(todayIso, -(2 * OVERVIEW_DAYS - 1));
  const counted = works.filter((w) => w.counted);
  const inNow = counted.filter((w) => dayOf(w.submittedAt) >= startNow && dayOf(w.submittedAt) <= todayIso);
  const inBefore = counted.filter((w) => dayOf(w.submittedAt) >= startBefore && dayOf(w.submittedAt) < startNow);
  const now = pct(inNow.map((w) => w.tally).reduce(add, ZERO));
  const before = pct(inBefore.map((w) => w.tally).reduce(add, ZERO));
  const waitingWorks = works.filter((w) => w.needsAdult);
  const points = counted
    .filter((w) => w.tally.decided > 0)
    .map((w): ChartPoint => ({ instanceId: w.instanceId, title: w.title, unitKey: w.unitKey, day: dayOf(w.submittedAt), pct: pct(w.tally) ?? 0, correct: w.tally.correct, decided: w.tally.decided }))
    .sort((a, b) => a.day.localeCompare(b.day) || a.instanceId.localeCompare(b.instanceId))
    .slice(-CHART_MAX);
  const judged = tags.filter((t) => t.tally.decided >= RESULT_RULES.minItems);
  const asc = [...judged].sort((a, b) => (ratio(a.tally) ?? 0) - (ratio(b.tally) ?? 0) || a.key.localeCompare(b.key));
  const weakest = asc.filter((t) => (ratio(t.tally) ?? 1) < RESULT_RULES.praiseFrom).slice(0, 3);
  const strongest = [...asc].reverse().filter((t) => !weakest.includes(t) && (ratio(t.tally) ?? 0) >= RESULT_RULES.praiseFrom).slice(0, 3);
  return {
    sent: { now: inNow.length, before: inBefore.length },
    firstTry: { now, before, diff: now !== null && before !== null ? now - before : null },
    waiting: { works: waitingWorks.length, oldestDays: waitingWorks.length ? Math.max(...waitingWorks.map((w) => w.waitingDays)) : null },
    main,
    points,
    weakest,
    strongest,
  };
}
```

```ts
export const otherRows = <R extends Pick<PendingRow, "group" | "kind">>(rows: readonly R[]): R[] => countedRows(rows).filter((r) => !isSubmissionRow(r));
export function otherView(p: { rows: readonly Pick<PendingRow, "group" | "kind">[]; failed: readonly unknown[]; more: boolean }): string | null {
  const failed = p.failed.some((f) => !(UNCOUNTED_SOURCES as readonly unknown[]).includes(f));

  return p.more ? `≥ ${n}` : String(n);
```
